            document.getElementById('balance-val').innerText = balance.toFixed(2);
            document.getElementById('streak-days').innerText = `${streak}-day streak`;

                document.getElementById('withdraw-status').innerText = "Eligible";

            streak += 1;
            balance += 0.10; // প্রতিদিন $0.10 USDT যোগ হবে
            localStorage.setItem('streak_days', streak);
            localStorage.setItem('usdt_balance', balance);
            updateUI();
            tg.showAlert("Daily reward claimed! +$0.10 USDT added.");
        }
